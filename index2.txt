    function addToCart(productId, size) {
      // Check stock first
      if (appData.cupsStock[size] <= 0) {
        alert(`Stok Cup ${size} habis! Silahkan hubungi Owner untuk menambah stok.`);
        return;
      }

      const prod = appData.products.find(p => p.id === productId);
      if (!prod) return;

      const price = prod.prices[size];
      const existing = cart.find(item => item.productId === productId && item.size === size);

      if (existing) {
        if (existing.qty + 1 > appData.cupsStock[size]) {
          alert(`Stok Cup ${size} tidak mencukupi untuk menambah item.`);
          return;
        }
        existing.qty++;
      } else {
        cart.push({
          productId,
          name: prod.name,
          size,
          price,
          qty: 1
        });
      }

      renderCart();
    }

      });

      updateCartTotals();
    }

    function updateCartQty(index, change) {
      const item = cart[index];
      if (!item) return;

      const newQty = item.qty + change;
      if (newQty <= 0) {
        cart.splice(index, 1);
      } else {
        // Check stock
        if (change > 0 && newQty > appData.cupsStock[item.size]) {
          alert(`Stok Cup ${item.size} terbatas! Sisa stok: ${appData.cupsStock[item.size]}`);
          return;
        }
        item.qty = newQty;
      }
      renderCart();
    }

    function removeCartItem(index) {
      cart.splice(index, 1);
      renderCart();
    }

    function updateCartTotals() {
      const total = cart.reduce((sum, i) => sum + (i.price * i.qty), 0);
      document.getElementById('cartSubtotal').innerText = `Rp ${total.toLocaleString()}`;
      document.getElementById('cartTotal').innerText = `Rp ${total.toLocaleString()}`;
      calculateChange();
    }

        document.getElementById('cashInput').value = total;
      } else {
        document.getElementById('cashInput').value = val;
      }
      calculateChange();
    }

    function calculateChange() {
      const total = cart.reduce((sum, i) => sum + (i.price * i.qty), 0);
      const cash = parseFloat(document.getElementById('cashInput').value) || 0;
      const change = cash - total;

      const display = document.getElementById('changeDisplay');
      if (change >= 0) {
        display.innerText = `Rp ${change.toLocaleString()}`;
        display.className = "font-mono text-sm font-bold text-emerald-400";
      } else {
        display.innerText = `- Rp ${Math.abs(change).toLocaleString()}`;
        display.className = "font-mono text-sm font-bold text-rose-400";
      }
    }

        return;
      }

      const total = cart.reduce((sum, i) => sum + (i.price * i.qty), 0);
      const cash = parseFloat(document.getElementById('cashInput').value) || 0;

      if (cash < total) {
        alert("Nominal pembayaran tunai kurang dari total!");
        return;
      }

      // Check all stock availability once more
      const neededStock = { S: 0, M: 0, L: 0 };
      cart.forEach(item => neededStock[item.size] += item.qty);

      for (let s in neededStock) {
        if (neededStock[s] > appData.cupsStock[s]) {
          alert(`Transaksi gagal! Stok Cup ${s} tidak mencukupi.`);
          return;
        }
      }

      // Deduct Cup Stocks
      for (let s in neededStock) {
        appData.cupsStock[s] -= neededStock[s];
      }

      // Create Transaction Record
      const newTx = {
        id: "TRX-" + Date.now().toString().slice(-6),
        timestamp: new Date().toISOString(),
        cashier: currentRole === 'owner' ? 'Owner' : 'Karyawan',
        items: [...cart],
        subtotal: total,
        total: total,
        cashPaid: cash,
        change: cash - total
      };

      appData.transactions.unshift(newTx);
      saveAppData();

      // Show Receipt Modal
      showReceiptModal(newTx);

      // Reset Cart & UI
      clearCart();
      renderStockTab();
      renderReportsTab();
    }

    /* ==========================================
       REPORTS & ANALYTICS DASHBOARD
    ========================================== */
    function renderReportsTab() {
      const txs = appData.transactions || [];

      // Calculate Stats
      const totalIncome = txs.reduce((sum, t) => sum + t.total, 0);
      const totalTx = txs.length;
      let totalCups = 0;
      const itemCounts = {};

      txs.forEach(t => {
        t.items.forEach(i => {
          totalCups += i.qty;
          itemCounts[i.name] = (itemCounts[i.name] || 0) + i.qty;
        });
      });

      let topProdName = "-";
      let maxQty = 0;
      for (let k in itemCounts) {
        if (itemCounts[k] > maxQty) {
          maxQty = itemCounts[k];
          topProdName = k;
        }
      }

      document.getElementById('statTotalIncome').innerText = `Rp ${totalIncome.toLocaleString()}`;
      document.getElementById('statTotalTx').innerText = `${totalTx} TRX`;
      document.getElementById('statTotalCups').innerText = `${totalCups} Cup`;
      document.getElementById('statTopProduct').innerText = topProdName;

      // Render History Table
      const historyBody = document.getElementById('historyTableBody');
      historyBody.innerHTML = '';

      if (txs.length === 0) {
        historyBody.innerHTML = `<tr><td colspan="7" class="p-6 text-center text-slate-500">Belum ada riwayat transaksi</td></tr>`;
      } else {
        txs.forEach((t, idx) => {
          const dateObj = new Date(t.timestamp);
          const tr = document.createElement('tr');
          tr.className = "hover:bg-slate-800/50 transition";
          
          const itemsDesc = t.items.map(i => `${i.name} (${i.size}) x${i.qty}`).join(', ');

          tr.innerHTML = `
            <td class="p-3.5 font-mono font-bold text-amber-400">#${t.id}</td>
            <td class="p-3.5 text-xs text-slate-400">${dateObj.toLocaleDateString('id-ID')} ${dateObj.toLocaleTimeString('id-ID', {hour:'2-digit', minute:'2-digit'})}</td>
            <td class="p-3.5"><span class="px-2 py-0.5 rounded bg-slate-800 text-[11px]">${t.cashier}</span></td>
            <td class="p-3.5 text-xs max-w-xs truncate text-slate-300" title="${itemsDesc}">${itemsDesc}</td>
            <td class="p-3.5 font-mono font-bold text-slate-100">Rp ${t.total.toLocaleString()}</td>
            <td class="p-3.5 text-center">
              <button onclick='showReceiptModal(${JSON.stringify(t)})' class="px-2 py-1 rounded bg-slate-800 hover:bg-slate-700 text-teal-300 text-xs">
                <i class="fa-solid fa-receipt"></i> Struk
              </button>
            </td>
            <td class="p-3.5 text-right">
              <button onclick="deleteTransaction(${idx})" class="owner-only-btn px-2 py-1 rounded bg-rose-500/20 text-rose-400 hover:bg-rose-500/30 text-xs">
                <i class="fa-solid fa-trash"></i>
              </button>
            </td>
          `;
          historyBody.appendChild(tr);
        });
      }

      // Apply permissions
      const isOwner = currentRole === 'owner';
      document.querySelectorAll('.owner-only-btn').forEach(b => b.style.display = isOwner ? '' : 'none');

      // Update Chart
      renderChart(txs);
    }

    function deleteTransaction(index) {
      if (currentRole !== 'owner') return;
      if (confirm("Hapus transaksi ini dari riwayat?")) {
        appData.transactions.splice(index, 1);
        saveAppData();
        renderReportsTab();
      }
    }

    function renderChart(txs) {
      const ctx = document.getElementById('salesChart').getContext('2d');

      // Group last 7 days sales
      const days = 7;
      const labels = [];
      const dataPoints = [];

      for (let i = days - 1; i >= 0; i--) {
        const d = new Date();
        d.setDate(d.getDate() - i);
        const dateStr = d.toLocaleDateString('id-ID', { day: 'numeric', month: 'short' });
        labels.push(dateStr);

        // Sum for this date
        const dayTotal = txs.filter(t => {
          const tDate = new Date(t.timestamp);
          return tDate.toDateString() === d.toDateString();
        }).reduce((sum, t) => sum + t.total, 0);

        dataPoints.push(dayTotal);
      }

      if (salesChartInstance) {
        salesChartInstance.destroy();
      }

      salesChartInstance = new Chart(ctx, {
        type: 'line',
        data: {
          labels: labels,
          datasets: [{
            label: 'Penjualan (Rp)',
            data: dataPoints,
            borderColor: '#F59E0B',
            backgroundColor: 'rgba(245, 158, 11, 0.15)',
            fill: true,
            tension: 0.4,
            borderWidth: 3,
            pointBackgroundColor: '#0D9488',
            pointRadius: 5
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: {
            legend: { display: false }
          },
          scales: {
            x: {
              grid: { color: 'rgba(255,255,255,0.05)' },
              ticks: { color: '#94a3b8' }
            },
            y: {
              grid: { color: 'rgba(255,255,255,0.05)' },
              ticks: { color: '#94a3b8' }
            }
          }
        }
      });
    }
